async function sendMessage() { const input = document.getElementById("userInput"); const chatBox = document.getElementById("chatBox"); if (!input || !input.value.trim()) return; const userText = input.value; input.value = ""; chatBox.innerHTML += `
You: ${userText}`; try { const apiKey = "YOUR_ACTUAL_API_KEY"; // Replace with your real key const url = `https://generativelanguage.googleapis.com/v1beta/models/gemini-3.5-flash-lite:generateContent?key=${apiKey}`; const response = await fetch(url, { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ contents: [{ parts: [{ text: userText }] }] }) }); const data = await response.json(); const reply = data.candidates[0].content.parts[0].text; chatBox.innerHTML += `
J.A.R.V.I.S.: ${reply}`; } catch (err) { console.error(err); chatBox.innerHTML += `
J.A.R.V.I.S.: Connection error.`; } } }
