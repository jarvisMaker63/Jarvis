async function sendMessage() { const inputElement = document.getElementById("userInput"); const chatBox = document.getElementById("chatBox"); if (!inputElement || !inputElement.value.trim()) return; const userText = inputElement.value; inputElement.value = ""; // Display what you typed chatBox.innerHTML += `
You: ${userText}`; try { const apiKey = "YOUR_ACTUAL_API_KEY"; // Replace with your key const url = `https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash:generateContent?key=${apiKey}`; const response = await fetch(url, { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ contents: [{ parts: [{ text: userText }] }] }) }); const data = await response.json(); // Extract the reply from the JSON response const reply = data.candidates[0].content.parts[0].text; // Display Jarvis's reply chatBox.innerHTML += `
J.A.R.V.I.S.: ${reply}`; } catch (err) { console.error(err); chatBox.innerHTML += `
J.A.R.V.I.S.: Connection error.`; } }
