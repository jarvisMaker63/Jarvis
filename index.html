async function sendMessage() { const input = document.getElementById("userinput"); const chatBox = document.getElementById("chatBox"); if (!input || !input.value.trim()) return; const userText = input.value; input.value = ""; chatBox.innerHTML += `
You: ${userText}
`; try { const apiKey = "YOUR_API_KEY_HERE"; const url = `https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash:generateContent?key=${apiKey}`; const response = await fetch(url, { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ contents: [{ parts: [{ text: userText }] }] }) }); const data = await response.json(); const reply = data.candidates[0].content.parts[0].text; chatBox.innerHTML += `
J.A.R.V.I.S.: ${reply}
`; } catch (err) { chatBox.innerHTML += `
J.A.R.V.I.S.: Connection error.
`; } }
